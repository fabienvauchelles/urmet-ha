#include <tunables/global>

# AppArmor profile for the Urmet doorphone gateway add-on. The profile name must
# match the add-on slug. It declares no Linux capability on purpose, grants
# read and memory-map over the read-only system (so Python can load its C
# extensions: pjsua2, av, aiortc), and confines writes to the add-on's own
# surfaces. No Home Assistant directory is mapped, so /config, /media, /share and
# /backup are simply not present in the container.
profile urmet_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # No capability lines: the gateway needs none beyond the container defaults.

  # The signals s6-overlay uses to supervise and stop the service tree.
  signal (send,receive) set=(int,term,kill,cont,hup),

  # SIP/TLS to the registrar, the cloud REST plane, and the WebRTC media leg.
  network,

  # s6-overlay, bashio, the shell and the coreutils the init scripts call.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /command/** ix,
  /package/** ix,
  /etc/s6-overlay/** ix,
  /run/{s6,s6-rc,service}/** ix,
  /usr/lib/bashio/** ix,

  # Read and memory-map the read-only system. mmap (m) is what lets the Python
  # interpreter load its shared-object extensions.
  /{usr/,usr/local/,}lib/** mr,
  /usr/lib/python3*/** mr,
  /usr/local/lib/python3*/** mr,
  /app/** mr,
  /etc/** r,
  /usr/share/** r,
  /proc/*/{fd,status,stat,cmdline}* r,

  # Writable surfaces: the add-on data dir, the process tmp, the s6 runtime.
  /data/** rwk,
  /tmp/** rwk,
  /run/** rwk,
  /var/run/** rwk,
  /dev/tty rw,
  /dev/null rw,
  /dev/urandom r,
}
